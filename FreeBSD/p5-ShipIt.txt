NAME
     shipit - Executable for ShipIt

SYNOPSIS
      $ shipit
      $ shipit --dry-run       # or -n
      $ shipit --write-config
      $ shipit --trial         # release -TRIAL version
      $ shipit --skip-tests    # but please don't.
