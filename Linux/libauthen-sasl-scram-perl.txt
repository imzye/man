NAME
       Authen::SASL::Perl::SCRAM_SHA_512 - SCRAM-SHA-512 support for Authen::SASL

VERSION
       0.04

SYNOPSIS
          # with Authen::SASL::SCRAM installed
          use Authen::SASL;

          my $client = Authen::SASL->new(
               username => 'user',
               password => 'pass',
               mechanism => 'SCRAM-SHA-512'
          );
          # authenticates using SCRAM SHA-512 hash
