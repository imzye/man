NAME
       XML::RPC::UA::LWP - XML::RPC useragent, using LWP

SYNOPSIS
           use XML::RPC::Fast;
           use XML::RPC::UA::LWP;

           my $rpc = XML::RPC::Fast->new(
               $uri,
               ua => XML::RPC::UA::LWP->new(
                   ua      => 'YourApp/0.1',
                   timeout => 3,
               ),
           );

DESCRIPTION
       Default syncronous useragent for XML::RPC::Fast

IMPLEMENTED METHODS
   new
   async = 0
   call
