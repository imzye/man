NAME
     ec_pnode_count - Count node occurrences in a parse subtree.

SYNOPSIS
     #include <ecoli/parse.h>

     unsigned int ec_pnode_count(
         /*
          * The node of the parsing tree where the search starts.
          */
         const struct ec_pnode   *root,
         const char              *id     /* The node identifier string to match. */
     );

RETURN VALUE
     The number of nodes matching the identifier.
