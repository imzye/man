NAME
       kdk_get_process_status - 获取某一进程的进程状态

SYNOPSIS
       #include <libkyprocess.h>

       extern char* kdk_get_process_status(int proc_num);

       Link with -lkyrtinfo.

Detailed Description
       The main function of the interface is to obtain the process state of the specified process.

       The argument proc_num is the PID of the process.

RETURN VALUE
       On success, returns the process state of the process of type char*; The returned string needs to be released by free

       On error, return NULL.

EXAMPLES
       #include "libkyprocess.h"
       #include <stdio.h>
       #include <stdlib.h>

       int main()
       {
           char *run_time = kdk_get_process_running_time(841);
           char *cpu_time = kdk_get_process_cpu_time(841);
           char *cmd = kdk_get_process_command(841);
           char *start_time = kdk_get_process_start_time(841);
           printf("获取某一进程的CPU利用率：%0.1f0, kdk_get_process_cpu_usage_percent(841));
           printf("获取某一进程的内存占用率：%0.1f0, kdk_get_process_mem_usage_percent(841));
           printf("获取某一进程的进程状态：%s0, kdk_get_process_status(841));
           printf("获取某一进程的进程端口号：%d0, kdk_get_process_port(841));
           printf("获取某一进程的启动时间：%s0, start_time);
           printf("获取某一进程的运行时间：%s0, run_time);
           printf("获取某一进程的CPU时间：%s0, cpu_time);
           printf("获取某一进程的Command：%s0, cmd);
           printf("获取某一进程的属主：%s0, kdk_get_process_user(841));
           free(run_time);
           free(cpu_time);
           free(cmd);
           free(start_time);
           char** pid =  kdk_procname_get_process_infomation("systemd");
           size_t index = 0;
           while (pid[index])
           {
               printf("pid %s0, pid[index]);
               index ++;
           }
           kdk_proc_freeall(pid);

           char** info = kdk_get_process_all_information();
           size_t count = 0;
           while (info[count])
           {
               printf("No. %d %s0, count + 1, info[count]);
               count ++;
           }
           kdk_proc_freeall(info);

           char *name = kdk_get_process_name(1151571);
           printf("name = %s0, name);
           free(name);

           int id =  kdk_get_process_id("systemd");
           printf("pid %d0, id);

           return 0;
       }

       kdk_get_process_all_information() and kdk_proc_freeall().
